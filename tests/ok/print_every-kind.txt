struct Point
{
    i32 mut x
    i32 mut y
    fn show := () -> i32
    {
        print("in member:")
        print(x)
        exit x
    }
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
}
struct Info
{
    bool ok
    i64 big
    Point at
}
fn make := (i32 v) -> Info
{
    print("in fn")
    Info i{v == 1, 5000000000, {v, v}}
    exit i
}
